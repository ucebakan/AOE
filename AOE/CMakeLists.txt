cmake_minimum_required(VERSION 3.25)
project(4UnityAOEManager VERSION 1.1.18 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "4UnityAOEManager requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
add_library(aoe_core STATIC src/common.cpp src/tracer.cpp src/analysis.cpp src/timing.cpp src/producer.cpp src/record_init.cpp src/caller_trace.cpp src/budget_write.cpp src/tick_patch.cpp src/initial_2x.cpp src/initial_nx_lifecycle.cpp src/operation_observer.cpp src/live_validation.cpp src/visual_suppression.cpp src/target_inspector.cpp src/selected_target_provenance.cpp src/target_writer_runtime.cpp src/settings.cpp src/profile.cpp src/manager_state.cpp src/ui_layout.cpp src/diagnostics.cpp)
target_include_directories(aoe_core PUBLIC src)
target_sources(aoe_core PRIVATE src/aoe_locator.cpp src/aoe_diagnostic.cpp)
target_link_libraries(aoe_core PUBLIC bcrypt)
add_executable(4UnityAOEManager WIN32 src/main.cpp src/app.rc)
target_link_libraries(4UnityAOEManager PRIVATE aoe_core comctl32 comdlg32 uxtheme dwmapi shell32 advapi32)
if(MSVC)
  target_link_options(4UnityAOEManager PRIVATE /MANIFEST:NO)
endif()
add_executable(trace_fixture tests/trace_fixture.cpp)
add_executable(tracer_tests tests/tests.cpp tests/timing_tests.cpp tests/producer_tests.cpp tests/record_init_tests.cpp tests/caller_trace_tests.cpp tests/budget_write_tests.cpp tests/tick_patch_tests.cpp tests/initial_2x_tests.cpp tests/initial_nx_lifecycle_tests.cpp tests/target_inspector_tests.cpp tests/selected_target_provenance_tests.cpp tests/target_writer_runtime_tests.cpp tests/live_validation_tests.cpp tests/visual_suppression_tests.cpp tests/manager_tests.cpp)
target_link_libraries(tracer_tests PRIVATE aoe_core)
add_dependencies(tracer_tests trace_fixture)
enable_testing()
add_executable(aoe_locator_tests tests/aoe_locator_tests.cpp)
target_link_libraries(aoe_locator_tests PRIVATE aoe_core)
add_test(NAME aoe_signature_relocation COMMAND aoe_locator_tests)
add_test(NAME tracer_unit_and_debugger COMMAND tracer_tests)
set_tests_properties(tracer_unit_and_debugger PROPERTIES TIMEOUT 120)
add_test(NAME gui_synthetic_smoke COMMAND 4UnityAOEManager --smoke-test)
set_tests_properties(gui_synthetic_smoke PROPERTIES TIMEOUT 30)
