cmake_minimum_required(VERSION 3.20)
project(PlayerCounter LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "PlayerCounter requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(player_counter_reader STATIC reader.cpp exit.cpp recovery.cpp)
target_compile_definitions(player_counter_reader PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(player_counter_reader PUBLIC /W4 /utf-8 /EHsc)
target_link_libraries(player_counter_reader PUBLIC bcrypt)
add_executable(PlayerCounter WIN32 main.cpp)
target_link_libraries(PlayerCounter PRIVATE player_counter_reader user32 gdi32)
target_link_options(PlayerCounter PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(PlayerCounterTests tests.cpp)
target_link_libraries(PlayerCounterTests PRIVATE player_counter_reader)
target_link_options(PlayerCounterTests PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
enable_testing()
add_test(NAME PlayerCounterTests COMMAND PlayerCounterTests)
add_executable(PlayerCounterUiTests WIN32 main.cpp)
target_compile_definitions(PlayerCounterUiTests PRIVATE PC_UI_TEST)
target_link_libraries(PlayerCounterUiTests PRIVATE player_counter_reader user32 gdi32)
target_link_options(PlayerCounterUiTests PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_test(NAME PlayerCounterUiTests COMMAND PlayerCounterUiTests)
