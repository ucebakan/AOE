cmake_minimum_required(VERSION 3.25)
project(UnityToolsNative LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(../../AOE aoe EXCLUDE_FROM_ALL)
add_subdirectory(../../PlayerCounter counter EXCLUDE_FROM_ALL)
add_library(UnityAoe SHARED ../../AOE/src/main.cpp)
target_compile_definitions(UnityAoe PRIVATE UNITY_SUITE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(UnityAoe PRIVATE /utf-8 /EHsc)
target_link_libraries(UnityAoe PRIVATE aoe_core comctl32 comdlg32 uxtheme dwmapi shell32 advapi32)
add_library(UnityCounter SHARED ../../PlayerCounter/main.cpp ../../PlayerCounter/tests.cpp count_monitor.cpp)
target_compile_definitions(UnityCounter PRIVATE UNITY_SUITE PC_TEST_EXPORT)
target_link_libraries(UnityCounter PRIVATE player_counter_reader user32 gdi32)
add_library(UnitySalesman SHARED salesman_bridge.cpp)
target_compile_definitions(UnitySalesman PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(UnitySalesman PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(UnitySalesman PRIVATE user32 bcrypt)
add_library(UnityCollection SHARED collection_bridge.cpp)
target_compile_definitions(UnityCollection PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(UnityCollection PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(UnityCollection PRIVATE user32 bcrypt)
